호스트 터미널에서 다음 명령어 실행(도커 데스크탑 내부 터미널에서 해도 무관)

# 1. postgresql 이미지 pull / volume 생성 / 컨테이너 생성
docker pull postgres:16

docker volume create robotdata3_sql

docker run -d --name robotdb3_sql \
-e POSTGRES_DB=robotdb3_sql \
-e POSTGRES_USER=rokey \
-e POSTGRES_PASSWORD=rokey \
-p 5432:5432 \
-v robotdata3_sql:/var/lib/postgresql/data \
postgres:16


# 2. redis 이미지 pull / volume 생성 / 컨테이너 생성
docker pull redis:8.8

docker volume create robotdata3_nosql

docker run -d --name robotdb3_nosql \
-p 6379:6379 \
-v robotdata3_nosql:/data \
redis:8.8 redis-server --appendonly yes \
--user rokey on ">rokey" ~* +@all
