cmake_minimum_required(VERSION 3.8)
project(hospital_dynamic_layer)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/prediction_layer.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} nav2_costmap_2d pluginlib rclcpp sensor_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
